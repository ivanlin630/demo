	return "task=%s reason=%s prio=%d opt=%s pos=%s mt=%s parent=%s parent_pos=%s parent_task=%s food=%.2f | tgt camp=%d outpost=%d owner=%d constr_team=%d" % [
		sub.current_task, sub.task_reason, sub.task_priority, sub.current_option, str(sub.tile_pos), str(sub.move_target),
		sub.parent_team_id, (str(p.tile_pos) if p else "-"), (p.current_task if p else "-"),
		float(sub.resources.get("food", 0.0)),
		(tt.camp_level if tt else -1), (tt.outpost_level if tt else -1), (tt.outpost_owner if tt else -9),
		(tt.construction_team_id if tt else -9)]


func _run(task: String, label: String, where: String) -> void:
	seed(SEED)
	var st := WorldState.new()
	GameSetup.setup(st, GameSetup.load_config("res://config/default.json"))
	var ptid: int = st.get_player_team_id()
	var pt: TeamData = st.teams[ptid]
	var tgt: Vector2i = pt.tile_pos if where == "腳下" else pt.tile_pos + Vector2i(1, 0)
	var leader: int = -1
	for pid in pt.named_members:
		if pid != pt.leader_id:
			leader = pid
			break
	st.player_state["sub_leader_id"] = leader
	st.player_state["sub_pop_count"] = 2
	st.player_state["sub_task"] = task
	st.player_state["sub_move_q"] = tgt.x
	st.player_state["sub_move_r"] = tgt.y
	print("---- [QA] %s｜%s  player_team=Team%d pos=%s tgt=%s" % [label, where, ptid, str(pt.tile_pos), str(tgt)])
	var r: Dictionary = PlayerCommandSystem.new().execute_action(st, -1, "dispatch_subteam")
	var sid: int = int(r.get("sub_id", -1))
	if sid == -1:
		print("[QA] 派不出去：%s" % String(r.get("msg", "")))
		return
	var sub: TeamData = st.teams[sid]
	var last: String = _snap(st, sub, tgt)
	print("[QA] t=%d Team%d 派出 %s" % [st.world.current_tick, sid, last])
	var runner := SimRunner.new()
	for i in range(WorldState.TICKS_PER_DAY):
		var p: TeamData = st.teams.get(st.get_player_team_id())
		runner.advance_tick(st, p.tile_pos if p != null else Vector2i(-1, -1))
		if st.encounter_active:
			runner._encounter_system.resolve_encounter_end(st, "draw")
		if not st.teams.has(sid):
			var pp: TeamData = st.teams.get(ptid)
			print("[QA] t=%d Team%d 消失（母隊 Team%d pop=%d 仍在=%s）；消失前最後狀態：%s" % [
				st.world.current_tick, sid, ptid, (pp.population if pp else -1), str(pp != null), last])
			return
		sub = st.teams[sid]
		var now: String = _snap(st, sub, tgt)
		if now.get_slice(" food=", 0) != last.get_slice(" food=", 0):
			print("[QA] t=%d Team%d %s" % [st.world.current_tick, sid, now])
		last = now
	print("[QA] t=%d Team%d 一天結束仍在：%s" % [st.world.current_tick, sid, last])
