def sub(p,old,new):

    o=old.replace('\n','\r\n') if '\r\n' in s else old
    nn=new.replace('\n','\r\n') if '\r\n' in s else new
    assert s.count(o)==1,(p,old[:40]); s=s.replace(o,nn); open(p,'w',encoding='utf-8',newline='').write(s)
