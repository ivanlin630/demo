# 迷霧系統 Implementation Plan（含偵查/潛行技能）

> **STATUS: FULLY IMPLEMENTED** — All Parts A–G complete as of 2026-06-02. This file is an archive of the original plan for reference only.
>
> **Implementation note:** `_avg_skill` in `vision_system.gd` uses `team.named_members` (not `team.advisors + team.members` as written in the plan below — those fields were removed during the `named_members` migration).

---

## Context

貿易系統完成，AI 目標選擇目前全知。現在統一實裝視野限制：
- 每個 team 只看到「已發現」的 team
- 視野範圍由**偵查技能**決定
- 能否看到目標 team 由**目標暴露值**（受**潛行技能**影響）決定
- 圖塊/據點：視野範圍內永遠可見；team：需通過暴露值檢定

---

## 核心公式

```
scout_range(obs) = round((VISION_RADIUS + avg_偵查(obs) × SCOUT_BONUS) × terrain_vision_mult(obs.tile))
exposure(target) = base_exposure(target.pop) × (1.0 - avg_潛行(target) × 0.6) × terrain_exposure_mult(target.tile)
dist_factor      = 1.0 - (dist / (scout_range + 1)) × 0.5   # 邊緣衰減至 0.5×
可見條件: exposure × dist_factor + avg_偵查(obs) × 0.3 > 0.5
```

**base_exposure 依人數（TEST VALUE）：**
```
base_exposure(pop) = clampf(0.2 + pop × 0.04, 0.2, 0.9)
# pop=5 → 0.40，pop=20 → 0.80，pop=50 → 0.90（上限）
```

**地形乘數：**
| terrain | terrain_vision_mult（觀察者） | terrain_exposure_mult（目標） |
|---|---|---|
| plains | 1.0 | 1.0 |
| forest | 0.6 | 0.5 |
| mountain | 0.8 | 0.7 |

- 觀察者在森林：視野縮短（樹木遮擋視線）
- 目標在森林：暴露值大幅降低（隱蔽）
- 兩者獨立計算

**技能屬性掛鉤：**
| 技能 | 主屬性 | 副屬性 | 成長公式 |
|---|---|---|---|
| 偵查 | 智力 | 體力 | `0.001 × 智力 × (0.5 + 體力 × 0.5) × skill_mult` |
| 潛行 | 體力 | 毅力 | `0.001 × 體力 × (0.5 + 毅力 × 0.5) × skill_mult` |

**成長觸發：**
- 偵查：每次偵查到新 team 時，observer team 全員成長
- 潛行：每次 observer 在視野內但未能偵測到 target 時，target team 全員成長

## Part B：VisionSystem（新建）

```gdscript
# scripts/simulation/vision_system.gd
class_name VisionSystem

const VISION_RADIUS: int  = 3    # TEST VALUE — 基礎視野 hex
const SCOUT_BONUS: float  = 2.0  # TEST VALUE — 偵查=1.0 最多 +2 hex

const TERRAIN_VISION_MULT: Dictionary  = { "plains": 1.0, "forest": 0.6, "mountain": 0.8 }
const TERRAIN_EXPOSURE_MULT: Dictionary = { "plains": 1.0, "forest": 0.5, "mountain": 0.7 }

func tick_discovery(state: WorldState, team_ids: Array) -> void:
    for tid in team_ids:
        if not state.team_discovered.has(tid):
            state.team_discovered[tid] = []
        var obs: TeamData  = state.teams[tid]
        var scout: float   = _avg_skill(state, obs, "偵查")
        var obs_tile       = _get_tile(state, obs.tile_pos)
        var obs_terrain    = obs_tile.terrain if obs_tile else "plains"
        var vmult: float   = float(TERRAIN_VISION_MULT.get(obs_terrain, 1.0))
        var vrange: int    = roundi((VISION_RADIUS + scout * SCOUT_BONUS) * vmult)
        for other_id in state.teams:
            if other_id == tid: continue
            var other: TeamData = state.teams[other_id]
            var dist: int = _hex_dist(obs.tile_pos, other.tile_pos)
            if dist > vrange: continue
            var exposure: float   = _get_exposure(state, other)
            var dist_f: float     = 1.0 - (float(dist) / float(vrange + 1)) * 0.5  # TEST VALUE
            var eff_exp: float    = exposure * dist_f
            if _can_detect(scout, eff_exp):
                var is_new: bool = not state.team_discovered[tid].has(other_id)
                _mark(state, tid, other_id)
                if is_new:
                    _grow_skill(state, obs, "偵查", "智力", "體力")
            else:
                _grow_skill(state, other, "潛行", "體力", "毅力")

func reveal_encounter(state: WorldState, id_a: int, id_b: int) -> void:
    _mark(state, id_a, id_b)
    _mark(state, id_b, id_a)

func _can_detect(scout: float, exposure: float) -> bool:
    return exposure + scout * 0.3 > 0.5

func _get_exposure(state: WorldState, team: TeamData) -> float:
    var base: float    = clampf(0.2 + team.population * 0.04, 0.2, 0.9)  # TEST VALUE
    var stealth: float = _avg_skill(state, team, "潛行")
    var tgt_tile       = _get_tile(state, team.tile_pos)
    var terrain        = tgt_tile.terrain if tgt_tile else "plains"
    var emult: float   = float(TERRAIN_EXPOSURE_MULT.get(terrain, 1.0))
    return base * (1.0 - stealth * 0.6) * emult

func _get_tile(state: WorldState, pos: Vector2i):
    return state.world.tiles.get(pos.x * 1000 + pos.y)

func _avg_skill(state: WorldState, team: TeamData, skill: String) -> float:
    # NOTE: implemented with named_members (not advisors/members)
    var total: float = 0.0; var count: int = 0
    for pid in ([team.leader_id] as Array) + team.named_members:
        var p: PersonData = state.persons.get(pid)
        if p: total += float(p.skills.get(skill, 0.0)); count += 1
    return total / maxf(count, 1)

func _grow_skill(state: WorldState, team: TeamData,
        skill: String, attr1: String, attr2: String) -> void:
    for pid in ([team.leader_id] as Array) + team.named_members:
        var p: PersonData = state.persons.get(pid)
        if p == null: continue
        var a1: float  = float(p.attributes.get(attr1, 0.5)) * p.get_attribute_mult(attr1)
        var a2: float  = float(p.attributes.get(attr2, 0.5)) * p.get_attribute_mult(attr2)
        var growth: float = 0.001 * a1 * (0.5 + a2 * 0.5) * p.get_skill_mult(skill)  # TEST VALUE
        p.skills[skill] = minf(float(p.skills.get(skill, 0.0)) + growth, 1.0)

func _mark(state: WorldState, obs_id: int, tgt_id: int) -> void:
    if not state.team_discovered.has(obs_id):
        state.team_discovered[obs_id] = []
    if not state.team_discovered[obs_id].has(tgt_id):
        state.team_discovered[obs_id].append(tgt_id)

func _hex_dist(a: Vector2i, b: Vector2i) -> int:
    var dx := b.x - a.x; var dy := b.y - a.y
    return (abs(dx) + abs(dx + dy) + abs(dy)) / 2
```

### E3. `_nearest_independent` — 改遍歷 discovered

```gdscript
func _nearest_independent(state: WorldState, from_team: TeamData) -> int:
    var best_id: int = -1; var best_d: int = 999
    for tid in state.team_discovered.get(from_team.team_id, []):
        if not state.teams.has(tid): continue
        var t: TeamData = state.teams[tid]
        if t.faction_id != -1 or t.team_id == from_team.team_id: continue
        var d: int = _hex_dist(from_team.tile_pos, t.tile_pos)
        if d < best_d: best_d = d; best_id = tid
    return best_id
```

### E4. `_find_trade_target` — 改遍歷 discovered

```gdscript
func _find_trade_target(state: WorldState, merchant: TeamData) -> int:
    var best_id: int = -1; var best_d: int = 999
    for tid in state.team_discovered.get(merchant.team_id, []):
        if tid == merchant.team_id or not state.teams.has(tid): continue
        var t: TeamData = state.teams[tid]
        if float(t.resources.get("coin", 0)) < TRADE_MIN_COIN: continue
        var d: int = _hex_dist(merchant.tile_pos, t.tile_pos)
        if d < best_d: best_d = d; best_id = tid
    return best_id
```
