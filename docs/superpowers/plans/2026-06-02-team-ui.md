```gdscript
extends Node

# Helper for rendering member list column
static func render_member_list_row(member: Dictionary, width: int) -> String:
	var role_str = "L" if member.role == "leader" else "M"
	var hp_pct = 0.0
	if member.hp_max > 0:
		hp_pct = (member.hp_current as float) / member.hp_max * 100.0
	
	var line = "[%s] %s loyalty:%.2f hp:%d%% stress:%.2f" % [
		role_str,
		member.name.substr(0, 15),
		member.loyalty,
		int(hp_pct),
		member.stress
	]
	
	return line.substr(0, width)

# Helper for rendering quick card
static func render_quick_card(member: Dictionary, width: int, height: int) -> Array:
	var lines = []
	lines.append("=" * width)
	lines.append("Name: %s" % member.name)
	lines.append("Role: %s" % ("Leader" if member.role == "leader" else "Member"))
	lines.append("Loyalty: %.2f | Fear: %.2f" % [member.loyalty, member.fear])
	lines.append("Stress: %.2f | HP: %d/%d" % [member.stress, member.hp_current, member.hp_max])
	lines.append("")
	
	# Top 3 skills by value
	var skills_sorted = member.values.keys().map(func(k): return {key=k, value=member.values[k]}).sort_custom(func(a,b): return a.value > b.value)
	skills_sorted = skills_sorted.slice(0, min(3, len(skills_sorted)))
	
	lines.append("Top Skills:")
	for skill in skills_sorted:
		lines.append("  %s: %.2f" % [skill.key, skill.value])
	
	lines.append("=" * width)
	
	# Pad to height
	while len(lines) < height:
		lines.append("")
	
	return lines.slice(0, height)

# Helper for rendering health details
static func render_health_detail(member: Dictionary, width: int, height: int) -> Array:
	var lines = []
	lines.append("=== HEALTH DETAIL ===")
	lines.append("%s (HP: %d/%d)" % [member.name, member.hp_current, member.hp_max])
	lines.append("")
	
	for part_name in member.body_parts.keys():
		var part = member.body_parts[part_name]
		var status = "OK" if part.hp == part.hp_max else ("DMG" if part.hp > part.hp_max * 0.3 else "CRIT")
		lines.append("  %s: %d/%d (%s)" % [part_name, part.hp, part.hp_max, status])
	
	lines.append("=" * width)
	
	while len(lines) < height:
		lines.append("")
	
	return lines.slice(0, height)

# Helper for rendering equipment
static func render_equipment(member: Dictionary, width: int, height: int) -> Array:
	var lines = []
	lines.append("=== EQUIPMENT ===")
	
	if member.equipped:
		for slot_name in member.equipped.keys():
			var item = member.equipped[slot_name]
			if item:
				lines.append("  [%s] %s" % [slot_name, item.get("name", "unknown")])
	
	lines.append("")
	lines.append("Inventory (%d):" % len(member.inventory))
	for item in member.inventory.slice(0, 5):
		lines.append("  - %s" % item.get("name", "unknown"))
	
	if len(member.inventory) > 5:
		lines.append("  ... +%d more" % (len(member.inventory) - 5))
	
	lines.append("=" * width)
	
	while len(lines) < height:
		lines.append("")
	
	return lines.slice(0, height)

# Helper for rendering aggregate stats
static func render_aggregate_stats(team_stats: Dictionary, width: int, height: int) -> Array:
	var lines = []
	lines.append("=== TEAM STATS ===")
	lines.append("Population: %d" % team_stats.population)
	lines.append("Avg Loyalty: %.2f (min: %.2f)" % [team_stats.avg_loyalty, team_stats.min_loyalty])
	if team_stats.risk_count_loyalty > 0:
		lines.append("  ⚠ At-risk: %d members" % team_stats.risk_count_loyalty)
	
	lines.append("Avg Stress: %.2f (max: %.2f)" % [team_stats.avg_stress, team_stats.max_stress])
	if team_stats.risk_count_stress > 0:
		lines.append("  ⚠ High stress: %d members" % team_stats.risk_count_stress)
	
	lines.append("Health: %.0f%% avg | wounded: %d | critical: %d" % [
		(team_stats.avg_hp / 100.0) * 100.0 if team_stats.avg_hp > 0 else 0,
		team_stats.wounded_count,
		team_stats.critical_count
	])
	
	lines.append("Carry: %.1f/%.1f kg" % [team_stats.total_weight, team_stats.carry_capacity])
	
	var days_left = 0.0
	if team_stats.food_consumption_per_day > 0:
		days_left = team_stats.food_qty / team_stats.food_consumption_per_day
	
	lines.append("Food: %d qty (~%.1f days)" % [team_stats.food_qty, days_left])
	
	lines.append("=" * width)
	
	while len(lines) < height:
		lines.append("")
	
	return lines.slice(0, height)
```

```gdscript
func _handle_member_selection(direction: int) -> void:
	if not _member_mode or not _member_snapshot:
		return
	
	var members = _member_snapshot.get("members_detail", [])
	if len(members) == 0:
		return
	
	_member_selection += direction
	_member_selection = clampi(_member_selection, 0, len(members) - 1)
	print("[Member Select] Now viewing %s" % members[_member_selection].name)

func _get_selected_member() -> Dictionary:
	if not _member_snapshot:
		return {}
	
	var members = _member_snapshot.get("members_detail", [])
	if _member_selection >= len(members):
		return {}
	
	return members[_member_selection]
```

```gdscript
func _render_member_mode_screen() -> String:
	if not _member_mode or not _member_snapshot:
		return ""
	
	# Terminal dimensions (approximate)
	var col_left_width = 25
	var col_center_width = 40
	var col_right_width = 30
	var height = 20
	
	# LEFT COLUMN: Member list
	var members = _member_snapshot.get("members_detail", [])
	var left_lines = []
	left_lines.append("=== MEMBERS ===")
	
	for i in range(len(members)):
		var prefix = "> " if i == _member_selection else "  "
		var line = prefix + team_ui_helper.render_member_list_row(members[i], col_left_width - 3)
		left_lines.append(line)
	
	while len(left_lines) < height:
		left_lines.append("")
	
	# CENTER COLUMN: Member detail (quick card, health, or equipment)
	var selected_member = _get_selected_member()
	var center_lines = []
	
	if selected_member:
		if _member_detail_submode == 0:
			center_lines = team_ui_helper.render_quick_card(selected_member, col_center_width, height)
		elif _member_detail_submode == 1:
			center_lines = team_ui_helper.render_health_detail(selected_member, col_center_width, height)
		elif _member_detail_submode == 2:
			center_lines = team_ui_helper.render_equipment(selected_member, col_center_width, height)
	
	while len(center_lines) < height:
		center_lines.append("")
	
	# RIGHT COLUMN: Aggregate stats
	var team_stats = _member_snapshot.get("team_stats", {})
	var right_lines = team_ui_helper.render_aggregate_stats(team_stats, col_right_width, height)
	
	# Combine three columns
	var result = ""
	for i in range(height):
		var left = left_lines[i].ljust(col_left_width) if i < len(left_lines) else " " * col_left_width
		var center = center_lines[i].ljust(col_center_width) if i < len(center_lines) else " " * col_center_width
		var right = right_lines[i].ljust(col_right_width) if i < len(right_lines) else " " * col_right_width
		result += left + center + right + "\n"
	
	return result
```

## Execution Notes

- **Test approach:** All tests use direct flag setting (e.g., `text_ui.set("_member_mode", true)`) rather than command_player() to avoid UI rendering issues in headless context. Actual game can use hotkey [M].
- **Caching:** Snapshot taken once at enter, not refreshed during session. No cache invalidation needed during member_mode (member data is static per session).
- **Boundary handling:** If player team has 0 members, member_mode entry is rejected with message "Cannot enter: no members".
- **Hotkey conflicts:** [M] is dual-use (toggle ON when off, toggle OFF when on). Submodes [D/E/T] are only active when member_mode is ON.
